\section*{Projects}

\begin{twocolentry}{Jun 2026 -- Sep 2026}
\textbf{TPU-Accelerated Analytical Query Processing}
\end{twocolentry}
\begin{onecolentry}
\textit{Visiting Student Research at Google Systems Research Group}
\begin{highlights}[parsep=0pt]
    \item \textbf{Problem:} Semantic queries combine relational processing with LLM inference, but execution across CPUs and AI accelerators incurs costly data movement. Running both on TPUs requires adapting data-dependent relational operations to a static execution model
    \item \textbf{Contribution:} NTA introduces TPU-native relational operators and hardware-aware physical planning in JAX/XLA, enabling the first execution of all 22 TPC-H queries entirely on TPUs and laying the groundwork for unified relational and semantic query processing
    \item \textbf{Impact:} Achieved geometric-mean speedups over DuckDB of $7.0\times$ at 100GB on one TPU and $68.6\times$ at 1TB on 16 TPUs, excluding compilation and data transfer
    \item \textbf{Outcome:} Submitted to VLDB 2027
\end{highlights}
\end{onecolentry}

\vspace{2mm}

\begin{twocolentry}{Jun 2025 -- Sep 2025}
\textbf{Intent-Driven Semantic Operator Optimization}
\end{twocolentry}
\begin{onecolentry}
\textit{Visiting Research with Prof. Yongjoo Park, UIUC}
\begin{highlights}[parsep=0pt]
    \item \textbf{Problem:} Treating semantic operators as monolithic model calls limits optimization, leaving substantial trade-offs between output quality, inference latency, and monetary cost
    \item \textbf{Contribution:} CADENZA compiles natural-language intent into task-specific operator DAGs, exposing intermediate results to relational optimization and jointly tuning physical implementations
    \item \textbf{Impact:} Reduced scenario-average latency and monetary cost by up to $165.7\times$ and $310.3\times$, respectively, against state-of-the-art systems on SemBench
    \item \textbf{Outcome:} Accepted at SIGMOD 2027 and VLDB 2026 (Demo)
\end{highlights}
\end{onecolentry}

\needspace{5\baselineskip}
\begin{twocolentry}{Jan 2023 -- Jun 2025}
\textbf{High-Performance Schemaless Graph Database System}
\end{twocolentry}
\begin{onecolentry}
\begin{highlights}[parsep=0pt]
    \item \textbf{Problem:} Schema heterogeneity in property graphs creates storage, execution, and optimization overheads that limit analytical query performance
    \item \textbf{Contribution:} TurboLynx integrates graphlet-based columnar storage, schema-aware query execution, and cost-based optimization to efficiently support both graph traversal and analytical workloads
    \item \textbf{Role:} Core system developer; authored approximately 50\% of commits in a 246K+ LOC codebase. Coded primarily without LLM assistance in 2023--2024
    \item \textbf{Impact:} Outperformed graph DBMSs by up to $183.9\times$ and relational DBMSs by up to $41.27\times$ across LDBC SNB Interactive, TPC-H, and DBpedia evaluations
    \item \textbf{Outcome:} Accepted at VLDB 2026 and VLDB 2026 (Demo)
    \item \textbf{Website:} \url{https://turbolynx.io/}
\end{highlights}
\end{onecolentry}

\vspace{2mm}

\begin{twocolentry}{Dec 2022 -- Dec 2023}
\textbf{High-Speed Dependency Graph Generation for Database Replay}
\end{twocolentry}
\begin{onecolentry}
\textit{Joint Research with SAP Labs Korea}
\begin{highlights}[parsep=0pt]
    \item \textbf{Problem:} Dependency graph generation bottlenecks database replay, limiting efficient testing of system changes on captured production workloads
    \item \textbf{Contribution:} DoppelGanger++ introduces a stateful single forward scan algorithm and its parallel extension to efficiently construct dependency graphs that preserve replay correctness
    \item \textbf{Impact:} Implemented in SAP HANA, bringing the research into a commercial database system
    \item \textbf{Outcome:} Accepted at SIGMOD 2024 and VLDB 2024 (Demo)
\end{highlights}
\end{onecolentry}

\vspace{2mm}

\begin{twocolentry}{Mar 2022 -- Nov 2022}
\textbf{Provably Efficient Join Sampling and Cardinality Estimation}
\end{twocolentry}
\begin{onecolentry}
\begin{highlights}[parsep=0pt]
    \item \textbf{Problem:} Generating training samples from relational data can require prohibitively large joins; existing sampling methods incurred additional input-size overheads for general joins
    \item \textbf{Contribution:} Degree-based rejection sampling (DRS) achieves $\widetilde{O}(\mathrm{AGM}/\mathrm{OUT})$ time for uniform sampling and cardinality estimation over arbitrary joins after linear-time index construction
    \item \textbf{Impact:} Resolves an open problem in join sampling and estimation by eliminating the extra input-size factor, supporting training-data generation without materializing full join results
    \item \textbf{Outcome:} Accepted at PODS 2023; the first PODS paper with authors affiliated with a Korean institution
\end{highlights}
\end{onecolentry}
